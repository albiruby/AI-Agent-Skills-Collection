import Toybox.Activity;
import Toybox.ActivityMonitor;
import Toybox.Application;
import Toybox.Graphics;
import Toybox.Lang;
import Toybox.System;
import Toybox.WatchUi;

class UltimateWatchFaceView extends WatchUi.WatchFace {
    private var _isAwake as Boolean = true;
    private var _requiresBurnIn as Boolean = false;
    private var _partialUpdatesAllowed as Boolean = true;
    private var _centerX as Number = 130;
    private var _centerY as Number = 130;

    public function initialize() {
        WatchFace.initialize();
        var devSettings = System.getDeviceSettings();
        if (devSettings has :requiresBurnInProtection) {
            _requiresBurnIn = devSettings.requiresBurnInProtection;
        }
        if (WatchUi.WatchFace has :onPartialUpdate) {
            _partialUpdatesAllowed = true;
        }
    }

    public function onLayout(dc as Dc) as Void {
        _centerX = dc.getWidth() / 2;
        _centerY = dc.getHeight() / 2;
    }

    public function onUpdate(dc as Dc) as Void {
        if (dc has :clearClip) {
            dc.clearClip();
        }
        dc.setColor(Graphics.COLOR_BLACK, Graphics.COLOR_BLACK);
        dc.clear();

        var clockTime = System.getClockTime();
        var timeStr = Lang.format("$1$:$2$", [
            clockTime.hour.format("%02d"),
            clockTime.min.format("%02d")
        ]);

        if (!_isAwake && _requiresBurnIn) {
            var shiftX = ((clockTime.min % 3) - 1) * 3;
            var shiftY = (((clockTime.min / 3) % 3) - 1) * 3;
            dc.setColor(Graphics.COLOR_LT_GRAY, Graphics.COLOR_TRANSPARENT);
            dc.drawText(
                _centerX + shiftX,
                _centerY + shiftY,
                Graphics.FONT_NUMBER_MEDIUM,
                timeStr,
                Graphics.TEXT_JUSTIFY_CENTER | Graphics.TEXT_JUSTIFY_VCENTER
            );
            return;
        }

        dc.setColor(Graphics.COLOR_WHITE, Graphics.COLOR_TRANSPARENT);
        dc.drawText(
            _centerX,
            _centerY - 12,
            Graphics.FONT_NUMBER_HOT,
            timeStr,
            Graphics.TEXT_JUSTIFY_CENTER | Graphics.TEXT_JUSTIFY_VCENTER
        );

        var hr = getSafeHeartRate();
        var hrText = (hr != null) ? (hr.toString() + " BPM") : "-- BPM";
        var hrColor = getStatusColorForHr(hr);
        dc.setColor(hrColor, Graphics.COLOR_TRANSPARENT);
        dc.drawText(
            _centerX,
            _centerY + 46,
            Graphics.FONT_XTINY,
            hrText,
            Graphics.TEXT_JUSTIFY_CENTER | Graphics.TEXT_JUSTIFY_VCENTER
        );

        if (_isAwake || (_partialUpdatesAllowed && !_requiresBurnIn)) {
            drawSecondsClipped(dc, clockTime.sec, false);
        }
    }

    public function onPartialUpdate(dc as Dc) as Void {
        if (_requiresBurnIn || !_partialUpdatesAllowed) {
            return;
        }
        var clockTime = System.getClockTime();
        drawSecondsClipped(dc, clockTime.sec, true);
    }

    private function drawSecondsClipped(dc as Dc, sec as Number, applyClip as Boolean) as Void {
        var secStr = sec.format("%02d");
        var font = Graphics.FONT_TINY;
        var dims = dc.getTextDimensions("88", font);
        var w = dims[0] + 6;
        var h = dims[1] + 4;
        var x = _centerX - (w / 2);
        var y = _centerY + 18;

        if (applyClip && (dc has :setClip)) {
            dc.setClip(x, y, w, h);
            dc.setColor(Graphics.COLOR_BLACK, Graphics.COLOR_BLACK);
            dc.fillRectangle(x, y, w, h);
        }
        dc.setColor(0x00E676, Graphics.COLOR_TRANSPARENT);
        dc.drawText(_centerX, y + (h / 2), font, secStr, Graphics.TEXT_JUSTIFY_CENTER | Graphics.TEXT_JUSTIFY_VCENTER);
    }

    private function getSafeHeartRate() as Number or Null {
        var actInfo = Activity.getActivityInfo();
        if (actInfo != null && actInfo.currentHeartRate != null) {
            return actInfo.currentHeartRate;
        }
        if (ActivityMonitor has :getHeartRateHistory) {
            var iter = ActivityMonitor.getHeartRateHistory(1, true);
            var sample = iter.next();
            if (sample != null && sample.heartRate != ActivityMonitor.INVALID_HR_SAMPLE) {
                return sample.heartRate;
            }
        }
        return null;
    }

    private function getStatusColorForHr(hr as Number or Null) as Number {
        if (hr == null || hr < 100) {
            return 0x00E676;
        } else if (hr <= 135) {
            return 0xFFB300;
        } else if (hr <= 160) {
            return 0xFF6D00;
        }
        return 0xFF1744;
    }

    public function onEnterSleep() as Void {
        _isAwake = false;
        WatchUi.requestUpdate();
    }

    public function onExitSleep() as Void {
        _isAwake = true;
        WatchUi.requestUpdate();
    }

    public function disablePartialUpdates() as Void {
        _partialUpdatesAllowed = false;
    }
}
